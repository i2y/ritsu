## 関数

関数は、エフェクトで三つに分かれます。total な関数は、宣言した範囲のどの入力にも答えます。実行時に確かめる関数には、検査が示せなかった場所があります。生成したコードはそこで値を確かめ、外れれば、その場所の失敗の種類（`rule` など）で失敗します。止まらないかもしれない関数は、再帰を持ちます。

#### `次`（`next`）

total な関数です。宣言した範囲のどの入力にも答えます。見出しにエフェクトは書いてありません。kaname が推論したものです。

```kn
fn 次(next)(注文(order): 注文, イベント(event): イベント) -> result[注文, string]
  match 注文
    カート c =>
      match イベント
        追加 a =>
          if c.明細.count() < 50
            ok(注文.カート { 明細: c.明細.append(a.明細) })
          else
            err("カートに入る明細は 50 件までです")
        削除 r => ok(注文.カート { 明細: c.明細.filter(一件 => 一件.品番 != r.品番) })
        確定 p =>
          if c.明細.count() = 0
            err("空のカートは確定できません")
          else
            ok(注文.確定済み { 明細: c.明細, 合計: 合計(明細: c.明細), 日付: p.日付 })
        取り消し x => ok(注文.取り消し済み { 理由: x.理由 })
        else => err("カートは先に確定します")
    確定済み p =>
      match イベント
        支払い y =>
          if y.金額 = p.合計
            ok(注文.支払い済み { 明細: p.明細, 日付: p.日付 })
          else
            err("金額が合計と違います")
        取り消し x => ok(注文.取り消し済み { 理由: x.理由 })
        else => err("確定済みの注文は、支払うか取り消します")
    支払い済み d =>
      match イベント
        発送 s => ok(注文.発送済み { 明細: d.明細, 追跡番号: s.追跡番号 })
        else => err("支払い済みの注文は発送します")
    発送済み, 取り消し済み => err("注文はもう閉じています")
```
